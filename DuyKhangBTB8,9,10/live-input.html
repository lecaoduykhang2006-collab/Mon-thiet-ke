Xem mã JavaScript cốt lõi (Core DOM Event)
// 1. Truy xuất các phần tử DOM const textInput = document.getElementById('user-input'); const liveSpan = document.getElementById('live-span'); const liveHeading = document.getElementById('live-heading'); // 2. Lắng nghe sự kiện 'input' (kích hoạt ngay khi người dùng gõ từng phím) textInput.addEventListener('input', function(event) { // Lấy giá trị hiện thời từ ô text const textValue = event.target.value; // Hiển thị ngay lập tức vào thẻ <span> bên cạnh liveSpan.textContent = textValue.trim() !== '' ? textValue : '(chưa nhập)'; // Hiển thị ngay lập tức vào thẻ <h2> bên dưới if (textValue.trim() !== '') { liveHeading.textContent = textValue; liveHeading.classList.remove('empty-state'); } else { liveHeading.textContent = 'Nội dung bạn nhập sẽ xuất hiện ngay lập tức tại đây...'; liveHeading.classList.add('empty-state'); } });
